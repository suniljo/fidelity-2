public class ProcessedTransaction {
    private String id;
    private Optional<String> status;
    public String getId() {
        return id;
    }
    public void setId(String id) {
        this.id = id;
    }
    public Optional<String> getStatus() {
        return status;
    }
    public void setStatus(Optional<String> status) {
        this.status = status;
    }
    public ProcessedTransaction(String id, Optional<String> status) {
        this.id = id;
        this.status = status;
    }
}

Stream<PendingTransaction> reconcile(Stream<PendingTransaction> pending, Stream<Stream<ProcessedTransaction>> processed) {
           List<ProcessedTransaction>processedTransactionStream = processed.flatMap(o->o).collect(Collectors.toList());
           List<Long> processedId = processedTransactionStream.stream().filter(o->o.getStatus().get().equalsIgnoreCase("DONE")).map(o->Long.parseLong(o.getId())).collect(Collectors.toList());
           return pending.filter(o->processedId.contains(o.getId()));
        }
